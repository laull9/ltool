cmake_minimum_required(VERSION 3.28)
project(ltool_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Catch2 3 REQUIRED)
add_subdirectory(.. ltool)
if(NOT LTOOL_PROCESS_TEST_ONLY)
    target_include_directories(ltool INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../pkgs/range-v3/include>")
endif()
add_executable(ltool_process_child process/child.cpp)
if(LTOOL_BUILD_MODULE)
    set(test_sources LModule_test.cpp)
elseif(LTOOL_PROCESS_TEST_ONLY)
    set(test_sources LProcess_test.cpp)
else()
    file(GLOB test_sources CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "*_test.cpp")
    list(REMOVE_ITEM test_sources LModule_test.cpp)
endif()
foreach(source IN LISTS test_sources)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_link_libraries(${name} PRIVATE Catch2::Catch2WithMain ltool::ltool)
    if(name STREQUAL "LModule_test")
        target_link_libraries(${name} PRIVATE ltool::module)
    elseif(name STREQUAL "LProcess_test")
        add_dependencies(${name} ltool_process_child)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 120)
endforeach()
