cmake_minimum_required(VERSION 3.16)
project(ltool_consumer LANGUAGES CXX)
if(LTOOL_SOURCE)
    add_subdirectory("${LTOOL_SOURCE}" ltool)
else()
    find_package(ltool CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_compile_features(consumer PRIVATE cxx_std_20)
target_link_libraries(consumer PRIVATE ltool::ltool)
enable_testing()
add_test(NAME consumer COMMAND consumer)
